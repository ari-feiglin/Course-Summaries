\def\newmathtype#1#2#3#4{%
    \expandafter\def\csname @#1\endcsname##1[##2]{\begingroup\left#2 ##1 \;\middle#3\; ##2 \right#4\endgroup}%
    \expandafter\def\csname @@#1\endcsname##1{\begingroup\left#2 ##1\right#4\endgroup}%
    \expandafter\def\csname #1\endcsname##1{\@ifnextchar[ {\csname @#1\endcsname{##1}}{\csname @@#1\endcsname{##1}}}%
}

\def\@mathify@col#1#2{%
    \relax\xdef#1{%
        #1%
        \ifx\@mathifyfstcol\@false%
            &%
        \fi%
        $#2$%
    }%
    \ifx\@mathifyfstcol\@true%
        \global\let\@mathifyfstcol\@false%
    \fi%
}

\def\@mathify@row#1{%
    \noalign{\xdef#1{%
        #1%
        \ifx\@mathifyfstrow\@false%
            \noexpand\cr%
        \fi%
    }%
    \ifx\@mathifyfstrow\@true%
        \global\let\@mathifyfstrow\@false%
    \fi%
    \global\let\@mathifyfstcol\@true}%
}

\def\mathify#1#2{%
    \def#1{}%
    \let\@mathifyfstrow\@true%
    \let\@mathifyfstcol\@true%
    \vbox{\everycr={\@mathify@row{#1}}\halign{&\@mathify@col{#1}{##}\cr
        #2\cr
    }}%
}

\def\commsqed#1#2#3#4#5{{%
    \mathify\@csq{#1}%
    \expandafter\drawdiagram\expanded{{\@csq}}{%
        \diagarrow{from={1,1}, to={1,2}, text={#2}, y distance=-.25cm}%
        \diagarrow{from={2,1}, to={2,2}, text={#3}, y distance=.25cm}%
        \diagarrow{from={1,1}, to={2,1}, text={#4}, x distance=-.25cm}%
        \diagarrow{from={1,2}, to={2,2}, text={#5}, x distance=-.25cm}%
    }%
}}

\def\commsq#1#2#3#4#5{%
    \par\medskip%
    \centerline{\commsqed{#1}{#2}{#3}{#4}{#5}}%
    \medskip%
}

\def\diagramed#1#2{{%
    \mathify\@csq{#1}%
    \expandafter\drawdiagram\expanded{{\@csq}}{#2}%
}}

\def\@diagram#1#2#3{%
    \par\medskip%
    \centerline{\diagramed{#2}{#3}#1}%
    \medskip%
}

\def\diagram{\@ifstar {\@diagram{.}}{\@diagram{}}}

\def\@multline#1{%
\hbox to\dimexpr\displaywidth-\leftskip-1cm\relax{%
    \ifx\@nfirst\space%
        $\displaystyle#1$\hfill%
        \global\let\@nfirst\undefined%
    \else%
        \hfill$\displaystyle#1$%
    \fi%
}}

\def\multline#1{%
    \let\@nfirst\space%
    \kern\leftskip\vcenter{\halign{\@multline{##}\crcr
        #1\crcr%
    }}%
}
